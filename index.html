-- Allow either participant in a direct message to READ its reactions. -- This is required so the sender can see reactions added by the recipient. -- Assumption: admin_messages.sender_id / recipient_id and -- admin_message_reactions.user_id contain the Supabase auth.uid() UUID. ALTER TABLE public.admin_message_reactions ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "Participants can view message reactions" ON public.admin_message_reactions; CREATE POLICY "Participants can view message reactions" ON public.admin_message_reactions FOR SELECT TO authenticated USING ( EXISTS ( SELECT 1 FROM public.admin_messages AS m WHERE m.id = admin_message_reactions.message_id AND ( m.sender_id = auth.uid() OR m.recipient_id = auth.uid() ) ) ); GRANT SELECT ON public.admin_message_reactions TO authenticated;
